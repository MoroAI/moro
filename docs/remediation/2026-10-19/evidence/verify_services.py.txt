import json, uuid, subprocess
from pathlib import Path
from datetime import datetime, timezone
import httpx
root=Path('/Users/mac/Dev/MoroAI/moro/.moro/dev/2026-10-19')
token=(root/'project/.moro/web.token').read_text().strip()
results={'time':datetime.now(timezone.utc).isoformat(),'checks':[]}
def check(name,response,expected):
    assert response.status_code == expected,(name,response.status_code,response.text[:250])
    results['checks'].append({'name':name,'status':response.status_code})
for label,port in [('native',8501),('alternate',8765),('compose',18501)]:
    base=f'http://127.0.0.1:{port}'
    with httpx.Client(base_url=base,timeout=40,trust_env=False) as c:
        check(label+' anonymous overview',c.get('/api/system/overview'),401)
        check(label+' login page',c.get('/login'),200)
        r=c.post('/login',data={'token':token},headers={'Origin':base})
        check(label+' login',r,303)
        assert 'httponly' in r.headers['set-cookie'].lower() and 'samesite=strict' in r.headers['set-cookie'].lower()
        check(label+' authenticated dashboard',c.get('/'),200)
        check(label+' authenticated overview',c.get('/api/system/overview'),200)
        check(label+' packaged asset',c.get('/static/vendor/alpine.min.js'),200)
        check(label+' cross origin',c.post('/api/operations/list_datasets',json={},headers={'Origin':'http://untrusted.invalid'}),403)
        for path in ['/api/terminal/execute','/api/system/terminal/execute']:
            check(label+' removed '+path,c.post(path,json={'command':'pwd'},headers={'Origin':base}),404)
        check(label+' untrusted host',c.get('/login',headers={'Host':'untrusted.invalid'}),403)
        def op(name,body):
            return c.post('/api/operations/'+name,json=body,headers={'Origin':base})
        check(label+' extra command rejected',op('list_datasets',{'command':'id'}),422)
        check(label+' traversal rejected',op('preview_dataset',{'dataset_id':'../../etc/passwd'}),422)
        check(label+' unknown operation',op('shell_exec',{}),404)
        r=op('list_datasets',{'limit':1}); check(label+' bounded list',r,200)
        assert len(r.json()['payload']['datasets'])==1
        r=op('compile_dataset',{'dataset_id':'ds_support'}); check(label+' real compile',r,200)
        version=r.json()['payload']['version_id']
        r=op('preview_dataset',{'dataset_id':version,'row_limit':1});check(label+' immutable preview',r,200)
        assert len(r.json()['payload']['sample_rows'])==1 and 'messages' in r.json()['payload']['sample_rows'][0]
        check(label+' missing training dataset',op('start_training',{'dataset_id':'missing'}),422)
        check(label+' no invented job',op('get_job_status',{'job_id':'missing'}),404)
        check(label+' models endpoint',c.get('/api/models/'),200)
        check(label+' no invented eval',c.get('/api/evaluation/results/missing'),404)
for label,port in [('native',8000),('compose',18000)]:
    with httpx.Client(base_url=f'http://127.0.0.1:{port}',timeout=70,trust_env=False) as c:
        check(label+' gateway anonymous',c.get('/v1/models'),401)
        c.headers['Authorization']='Bearer '+token
        r=c.get('/v1/models'); check(label+' real model inventory',r,200)
        results['checks'][-1]['model_count']=len(r.json()['data'])
        r=c.post('/v1/chat/completions',json={'model':'moro-smoke-missing-model','messages':[{'role':'user','content':'hello'}]})
        check(label+' missing backend model',r,502)
        assert 'choices' not in r.json()
for label,port in [('native',8001),('cli-http',8002),('compose',18001)]:
    with httpx.Client(base_url=f'http://127.0.0.1:{port}',timeout=20,trust_env=False) as c:
        r=c.get('/health'); check(label+' webhook health',r,200)
        assert set(r.json())=={'status','service'}
        payload={'session_id':'verification-'+uuid.uuid4().hex,'prompt':'Synthetic verification prompt','completion':'Synthetic verification response','human_correction':'Synthetic reviewed response','user_rating':1}
        check(label+' webhook unauthenticated',c.post('/webhook/ingest',json=payload),403)
        c.headers.update({'X-API-Key':token,'Idempotency-Key':'verification-'+uuid.uuid4().hex})
        r=c.post('/webhook/ingest',json=payload); check(label+' feedback accepted',r,200)
        first=r.json()
        r=c.post('/webhook/ingest',json=payload); check(label+' replay',r,200)
        assert first==r.json() and first['count']==1
        payload['completion']='Conflicting synthetic response'
        check(label+' conflicting replay',c.post('/webhook/ingest',json=payload),409)
        payload['user_rating']=5
        check(label+' invalid rating',c.post('/webhook/ingest',json=payload),422)
        check(label+' browser origin',c.post('/webhook/ingest',json=payload,headers={'Origin':'http://127.0.0.1:8501'}),403)
with httpx.Client(base_url='http://127.0.0.1:8501',headers={'Authorization':'Bearer '+token},timeout=40,trust_env=False) as c:
    r=c.get('/api/flywheel/status');check('feedback actual status',r,200)
    before=r.json()['total_pairs_generated']
    r=c.post('/api/flywheel/cycle',json={});check('feedback actual extraction',r,200)
    count=r.json()['pairs_generated']
    r=c.get('/api/flywheel/pairs'); check('feedback actual pairs',r,200)
    assert r.json()['total']==before+count
    assert all(p['source_log_id'] for p in r.json()['pairs'])
    r=c.get('/api/system/overview');check('overview same feedback database',r,200)
    assert r.json()['flywheel']['pairs_generated']==before+count
results['containers']=[]
for name in ['moro-remediation-dashboard-1','moro-remediation-gateway-1','moro-remediation-webhook-1','moro-remediation-ollama-1','moro-remediation-flywheel-1','moro-audit-20261002-cpu-dev','moro-audit-20261002-ollama']:
    d=json.loads(subprocess.check_output(['docker','inspect',name]))[0]
    results['containers'].append({'name':name,'image':d['Image'],'user':d['Config']['User'],'ports':d['NetworkSettings']['Ports'],'running':d['State']['Running'],'health':d['State'].get('Health',{}).get('Status')})
    assert d['State']['Running'],name
(root/'services-smoke.json').write_text(json.dumps(results,indent=2)+'\n')
print(f"Passed {len(results['checks'])} live HTTP checks; all seven owned containers are running. No credentials recorded.")
